import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split, KFold, RandomizedSearchCV, cross_val_score
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score, silhouette_score
from sklearn.neural_network import MLPRegressor
from sklearn.decomposition import PCA
from kmodes.kprototypes import KPrototypes  # <-- New import for mixed data clustering
from scipy.stats import randint
import matplotlib.pyplot as plt
import seaborn as sns
import lightgbm as lgb
import xgboost as xgb
from catboost import CatBoostRegressor
import warnings
warnings.filterwarnings('ignore')

# Set global font family and size
plt.rc('font', family='DejaVu Serif')
plt.rc('font', size=14)

# --- 1) File path ---
# Please update this file path to where your data is located.
data_filepath = r"/content/WEKA-5-Goesyntetic.csv"

# --- 2) Load raw strain–stress data ---
print("\n=== Loading strain-stress data ===")
df_raw = pd.read_csv(data_filepath, header=None)

# --- 3) Extract metadata and strain ---
metadata_row = df_raw.iloc[0, 1:]
strain = pd.to_numeric(df_raw.iloc[2:, 0], errors='coerce')

# --- 4) Parse stress columns + metadata ---
dfs = []
test_id = 1
for i, col_data in enumerate(metadata_row):
    if isinstance(col_data, str) and pd.notna(col_data):
        parts = col_data.split(',')
        S = float(parts[0].split('=')[1])
        G = float(parts[1].split('=')[1])
        OMC = float(parts[2].split('=')[1])
        NS = float(parts[3].split('=')[1])
        stress_values = pd.to_numeric(df_raw.iloc[2:, i + 1], errors='coerce')
        df_temp = pd.DataFrame({
            'Strain': strain,
            'Stress_kPa': stress_values,
            'S': S,
            'G': G,
            'OMC': OMC,
            'NS': NS,
            'Test': test_id
        })
        dfs.append(df_temp)
        test_id += 1

# --- 5) Combine & clean ---
df = pd.concat(dfs, ignore_index=True)
df = df.dropna(subset=['Stress_kPa'])
print("\n=== Combined tidy data ===")
print(df.head())

# --- 6) Prepare X and y ---
features = ['S', 'G', 'OMC', 'NS', 'Strain']
X = df[features]
y = df['Stress_kPa']

# --- 7) Scaling Strategy for K-Prototypes & Regression ---
num_cols = ['OMC', 'NS', 'Strain']
cat_cols = ['S', 'G']

# We still need a fully scaled X for the regression models later
scaler_full = StandardScaler()
X_scaled = scaler_full.fit_transform(X)

# Create a mixed dataset for clustering
X_mixed = X.copy()
scaler_kproto = StandardScaler()
X_mixed[num_cols] = scaler_kproto.fit_transform(X_mixed[num_cols])

#------------------------------------------------------------------------------------------
# --- OPTIMAL CLUSTER EVALUATION (Elbow Method & Silhouette Score) ---
print("\n=== Evaluating Optimal Number of Clusters (k) ===")

X_for_pca = pd.get_dummies(X, columns=['S', 'G'])
X_for_pca[num_cols] = scaler_kproto.transform(X[num_cols]) 

costs = []
silhouette_scores = []
K_range = range(2, 11) # Testing k from 2 to 10
categorical_indices = [0, 1]
X_mixed_matrix = X_mixed.values

print("Running K-Prototypes for k=2 to 10 (this may take a moment)...")
for k in K_range:
    kproto_eval = KPrototypes(n_clusters=k, init='Cao', n_init=5, verbose=0, random_state=42)
    cluster_labels = kproto_eval.fit_predict(X_mixed_matrix, categorical=categorical_indices)
    costs.append(kproto_eval.cost_)
    sil_score = silhouette_score(X_for_pca, cluster_labels, metric='euclidean')
    silhouette_scores.append(sil_score)
    print(f"k={k} | Cost (Elbow): {kproto_eval.cost_:.2f} | Silhouette Score: {sil_score:.4f}")

# --- Plotting the Results ---
fig, ax1 = plt.subplots(figsize=(10, 6))
color = 'tab:blue'
ax1.set_xlabel('Number of Clusters (k)', fontsize=14)
ax1.set_ylabel('Cost (Elbow Method)', color=color, fontsize=14)
ax1.plot(K_range, costs, marker='o', color=color, linewidth=2, markersize=8, label='Cost (Lower is better)')
ax1.tick_params(axis='y', labelcolor=color)
ax1.set_xticks(K_range)

ax2 = ax1.twinx()  
color = 'tab:red'
ax2.set_ylabel('Silhouette Score', color=color, fontsize=14)
ax2.plot(K_range, silhouette_scores, marker='s', color=color, linewidth=2, markersize=8, label='Silhouette (Higher is better)')
ax2.tick_params(axis='y', labelcolor=color)

plt.title('Optimal Cluster Selection: Elbow Method vs. Silhouette Score', fontsize=16)
fig.tight_layout() 
plt.grid(True, linestyle='--', alpha=0.5)
plt.savefig(r"Optimal_K_Evaluation.png", dpi=600)
plt.show()

#----------------------------------------------------------------------------------------

# --- 7.5) STREAM 2: Apply K-Prototypes Clustering with Optimal k=4 ---
print("\n=== Profiling the 4 Interface Archetypes ===")

kproto_optimal = KPrototypes(n_clusters=4, init='Cao', n_init=5, verbose=0, random_state=42)
categorical_indices = [0, 1]  # S and G
X_mixed_matrix = X_mixed.values
df['Cluster'] = kproto_optimal.fit_predict(X_mixed_matrix, categorical=categorical_indices)

soil_map = {1.0: 'S1 (Sandy/Coarser)', 2.0: 'S2 (Clayey/Finer)'}
geo_map = {
    0.0: 'Natural Soil (G=0)', 
    1.0: 'Geogrid (G=1)', 
    2.0: 'Geotextile (G=2)', 
    3.0: 'Geocomposite 1 (G=3)', 
    4.0: 'Geocomposite 2 (G=4)', 
    5.0: 'GTSand (G=5)'
}

df['Soil_Name'] = df['S'].map(soil_map)
df['Geo_Name'] = df['G'].map(geo_map)

profile_list = []
for cluster_id in range(4):
    cluster_data = df[df['Cluster'] == cluster_id]
    profile = {
        'Cluster': f'Archetype {cluster_id + 1}',
        'Size (N)': len(cluster_data),
        'Mean Stress (kPa)': round(cluster_data['Stress_kPa'].mean(), 2),
        'Mean Strain (%)': round(cluster_data['Strain'].mean(), 2),
        'Mean Normal Stress (kPa)': round(cluster_data['NS'].mean(), 2),
        'Mean Moisture (+OMC%)': round(cluster_data['OMC'].mean(), 2),
        'Dominant Soil': cluster_data['Soil_Name'].mode().iloc[0],
        'Dominant Geosynthetic': cluster_data['Geo_Name'].mode().iloc[0]
    }
    profile_list.append(profile)

profile_df = pd.DataFrame(profile_list)
pd.set_option('display.max_columns', None)
pd.set_option('display.width', 1000)

print("\nCluster Profiles (Real-World Physical Characteristics):")
print("-" * 120)
print(profile_df.to_string(index=False))
print("-" * 120)

# --- 6. Visualize the 4 Archetypes using a Profile Heatmap ---
# Extract only the numerical columns for the heatmap
heatmap_data = profile_df[['Mean Stress (kPa)', 'Mean Strain (%)', 'Mean Normal Stress (kPa)', 'Mean Moisture (+OMC%)']].copy()
heatmap_data.index = profile_df['Cluster']

# Normalize the data column-wise (Min-Max scaling from 0 to 1) purely for the color mapping
# This ensures that variables with small scales (Moisture) and large scales (Stress) can share the same colormap
heatmap_colors_normalized = (heatmap_data - heatmap_data.min()) / (heatmap_data.max() - heatmap_data.min())

plt.figure(figsize=(10, 6))
sns.heatmap(
    heatmap_colors_normalized,    # Use normalized data for the background color
    annot=heatmap_data,           # Use raw, real-world data for the text numbers
    fmt='.2f',                    # Format numbers to 2 decimal places
    cmap='Pastel2',                # Color map (Yellow to Green to Blue)
    cbar_kws={'label': 'Relative Feature Intensity (Min to Max)'},
    linewidths=1, 
    linecolor='black',
    annot_kws={'size': 14, 'weight': 'bold'}
)
plt.title('Interface Archetypes Heatmap: Physical Profiles', fontsize=18, pad=20)
plt.xlabel('Physical Features', fontsize=16, labelpad=10)
plt.ylabel('Cluster Archetypes', fontsize=16, labelpad=10)
plt.xticks(rotation=15, ha='right', fontsize=12)
plt.yticks(rotation=0, fontsize=14)
plt.tight_layout()
plt.savefig(r"KPrototypes_Archetypes_Heatmap.png", dpi=600)
plt.show()

#----------------------------------------------------------------------------------------

# --- 8) Split: 70% train, 30% temp ---
X_train, X_temp, y_train, y_temp = train_test_split(
    X_scaled, y, test_size=0.30, random_state=42
)

# --- 9) Split temp: 50% val, 50% test → each 15% of original ---
X_val, X_final_test, y_val, y_final_test = train_test_split(
    X_temp, y_temp, test_size=0.50, random_state=42
)

X_train_val = np.vstack((X_train, X_val))
y_train_val = pd.concat([y_train, y_val])
print(f"\nTrain + Validation Set: {X_train_val.shape}")
print(f"Final Test Set: {X_final_test.shape}")

# Define regression models and their hyperparameter distributions
models = {
    'LightGBM': {
        'estimator': lgb.LGBMRegressor(random_state=42, verbose=-1),
        'param_dist': {
            'num_leaves': randint(20, 100),
            'learning_rate': [0.01, 0.05, 0.1],
            'n_estimators': randint(100, 500),
            'reg_alpha': [0, 0.1, 0.5, 1],
            'reg_lambda': [0, 0.1, 0.5, 1]
        }
    },
    'MLP': {
        'estimator': MLPRegressor(random_state=42, max_iter=500, early_stopping=True),
        'param_dist': {
            'hidden_layer_sizes': [(50,), (100,), (50, 50), (100, 50)],
            'activation': ['tanh', 'relu'],
            'alpha': [0.0001, 0.001, 0.01],
            'learning_rate': ['constant', 'adaptive']
        }
    },
    'XGBoost': {
        'estimator': xgb.XGBRegressor(random_state=42, verbosity=0),
        'param_dist': {
            'max_depth': randint(3, 10),
            'learning_rate': [0.01, 0.05, 0.1],
            'n_estimators': randint(100, 500),
            'reg_alpha': [0, 0.1, 0.5, 1],
            'reg_lambda': [0, 0.1, 0.5, 1]
        }
    },
    'CatBoost': {
        'estimator': CatBoostRegressor(random_state=42, verbose=0),
        'param_dist': {
            'depth': randint(3, 10),
            'learning_rate': [0.01, 0.05, 0.1],
            'iterations': randint(100, 500),
            'l2_leaf_reg': [1, 3, 5, 7, 9]
        }
    }
}

model_colors = {}
for model_name in models:
    if model_name == 'LightGBM':
        base_color = 'blue'
    elif model_name == 'MLP':
        base_color = 'orange'
    elif model_name == 'XGBoost':
        base_color = 'red'
    elif model_name == 'CatBoost':
        base_color = 'green'
    model_colors[model_name] = sns.light_palette(base_color, n_colors=6)[3]

kf = KFold(n_splits=5, shuffle=True, random_state=42)
results = {}

# --- 10) Hyperparameter tuning and evaluation for each regression model ---
for model_name, model_data in models.items():
    print(f"\n=== Hyperparameter Tuning for {model_name} ===")
    estimator = model_data['estimator']
    param_dist = model_data['param_dist']
    
    rand_search = RandomizedSearchCV(
        estimator,
        param_distributions=param_dist,
        n_iter=10,
        cv=kf,
        scoring='neg_mean_squared_error',
        random_state=42,
        n_jobs=-1
    )
    rand_search.fit(X_train_val, y_train_val)
    
    best_params = rand_search.best_params_
    print(f"Best Hyperparameters for {model_name}: {best_params}")
    
    # --- 11) K-Fold Cross-Validation ---
    best_cv_model = estimator.__class__(**best_params, random_state=42)
    if model_name == 'LightGBM':
        best_cv_model.set_params(verbose=-1)
    elif model_name == 'XGBoost':
        best_cv_model.set_params(verbosity=0)
    elif model_name == 'CatBoost':
        best_cv_model.set_params(verbose=0)
    elif model_name == 'MLP':
        best_cv_model.set_params(max_iter=500, early_stopping=True)
    
    r2_scores = cross_val_score(best_cv_model, X_train_val, y_train_val, cv=kf, scoring='r2')
    rmse_scores = np.sqrt(-cross_val_score(best_cv_model, X_train_val, y_train_val, cv=kf, scoring='neg_mean_squared_error'))
    print(f"Average Cross-Validation R²: {r2_scores.mean():.4f} ± {r2_scores.std():.4f}")
    
    # --- 12) Train ensemble for uncertainty ---
    n_ensemble = 10
    ensemble = []
    for i in range(n_ensemble):
        model = estimator.__class__(**best_params, random_state=42 + i)
        if model_name == 'LightGBM':
            model.set_params(verbose=-1)
        elif model_name == 'XGBoost':
            model.set_params(verbosity=0)
        elif model_name == 'CatBoost':
            model.set_params(verbose=0)
        elif model_name == 'MLP':
            model.set_params(max_iter=500, early_stopping=True)
        model.fit(X_train_val, y_train_val)
        ensemble.append(model)
    
    # --- 13) Predict on final test set ---
    all_preds = np.array([m.predict(X_final_test) for m in ensemble])
    y_pred = np.mean(all_preds, axis=0)
    y_std = np.std(all_preds, axis=0)
    lower = y_pred - 1.96 * y_std
    upper = y_pred + 1.96 * y_std
    
    mae = mean_absolute_error(y_final_test, y_pred)
    rmse = np.sqrt(mean_squared_error(y_final_test, y_pred))
    r2 = r2_score(y_final_test, y_pred)
    
    print(f"Final Unseen Test Metrics -> RMSE: {rmse:.4f} | R2: {r2:.4f}")
    
    results[model_name] = {
        'y_pred': y_pred,
        'lower': lower,
        'upper': upper,
        'mae': mae,
        'rmse': rmse,
        'r2': r2
    }

# --- 14) Actual vs Predicted plots in 2x2 grid ---
fig, axs = plt.subplots(2, 2, figsize=(16, 16))
axs = axs.flatten()
y_final_test_reset = y_final_test.reset_index(drop=True)

for i, model_name in enumerate(models.keys()):
    ax = axs[i]
    res = results[model_name]
    y_pred = res['y_pred']
    mae = res['mae']
    rmse = res['rmse']
    r2 = res['r2']
    
    ax.scatter(
        y_final_test_reset, y_pred,
        color=model_colors[model_name], edgecolor='black', alpha=0.7, s=50, label='Predictions'
    )
    min_val = min(y_final_test_reset.min(), y_pred.min())
    max_val = max(y_final_test_reset.max(), y_pred.max())
    padding = (max_val - min_val) * 0.05
    ax.plot(
        [min_val - padding, max_val + padding],
        [min_val - padding, max_val + padding],
        'r--', lw=2, label='Ideal Fit (y = x)'
    )
    ax.set_xlim(min_val - padding, max_val + padding)
    ax.set_ylim(min_val - padding, max_val + padding)
    ax.set_aspect('equal', adjustable='box')
    ax.set_xlabel('Actual Stress (kPa)', fontsize=16)
    ax.set_ylabel('Predicted Stress (kPa)', fontsize=16)
    ax.set_title(f'Actual vs. Predicted ({model_name})', fontsize=18)
    ax.text(
        0.05, 0.95,
        f'RMSE: {rmse:.4f} kPa\nMAE: {mae:.4f} kPa\nR²: {r2:.4f}',
        transform=ax.transAxes,
        fontsize=14, verticalalignment='top',
        bbox=dict(boxstyle='round,pad=0.5', facecolor='wheat', alpha=0.5)
    )
    ax.legend(prop={'size': 14}, loc='lower right')
    ax.grid(True, linestyle='--', color='gray', linewidth=0.5, alpha=0.5)
    ax.tick_params(axis='both', which='major', labelsize=14)

plt.tight_layout()
plt.savefig(r"Actual_vs_Predicted_All_Models.png", dpi=600)
plt.show()

print("\n=== Finished ===")
